function updateCountdown() {
  const now = new Date();
  const upcoming = tripData.events.filter(e => new Date(e.ts) > now);
  const card = document.getElementById('countdownCard');
  if (upcoming.length === 0) {
    document.getElementById('cdEventTitle').textContent = "旅行結束，平安回家 🏠";
    document.getElementById('cdTimer').style.display = 'none';
    document.getElementById('cdEventTime').textContent = '';
    card.classList.add('done');
    return;
  }
  const next = upcoming[0];
  const target = new Date(next.ts);
  const diff = target - now;
  const days = Math.floor(diff / 86400000);
  const hours = Math.floor((diff % 86400000) / 3600000);
  const mins = Math.floor((diff % 3600000) / 60000);
  const secs = Math.floor((diff % 60000) / 1000);
  document.getElementById('cdEventTitle').textContent = next.title;
  document.getElementById('cdDays').textContent = String(days).padStart(2, '0');
  document.getElementById('cdHours').textContent = String(hours).padStart(2, '0');
  document.getElementById('cdMins').textContent = String(mins).padStart(2, '0');
  document.getElementById('cdSecs').textContent = String(secs).padStart(2, '0');
  const jstTime = target.toLocaleString('zh-TW', { timeZone: 'Asia/Tokyo', month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit', timeZoneName: 'short' });
  const localTime = target.toLocaleString('zh-TW', { month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit' });
  document.getElementById('cdEventTime').textContent = '日本時間 ' + jstTime + '　你的時間 ' + localTime;
}
setInterval(updateCountdown, 1000);
updateCountdown();

function mapsLink(lat, lng) { return 'https://www.google.com/maps/search/?api=1&query=' + lat + ',' + lng; }
function navLink(points) {
  if (points.length < 2) return '#';
  const origin = points[0];
  const dest = points[points.length - 1];
  const waypoints = points.slice(1, -1).map(p => p.lat + ',' + p.lng).join('|');
  let url = 'https://www.google.com/maps/dir/?api=1&origin=' + origin.lat + ',' + origin.lng + '&destination=' + dest.lat + ',' + dest.lng + '&travelmode=transit';
  if (waypoints) url += '&waypoints=' + waypoints;
  return url;
}
function embedMap(points) {
  if (points.length === 0) return '';
  const markers = points.map(p => p.lat + ',' + p.lng).join('%7C');
  return 'https://maps.google.com/maps?q=' + markers + '&z=12&output=embed';
}
